changing the send realtime input:
src\lib\genai-live-client.ts

inside the sendRealtiemInput function... replace...

this:
for (const ch of chunks) {
  this.session?.sendRealtimeInput({ media: ch });
  if (ch.mimeType.includes("audio")) {
    hasAudio = true;
  }
  if (ch.mimeType.includes("image")) {
    hasVideo = true;
  }
  if (hasAudio && hasVideo) {
    break;
  }
}

with this:
for (const ch of chunks) {
  if (ch.mimeType.includes("audio")) {
    this.session?.sendRealtimeInput({ audio: ch });
    hasAudio = true;
  } else if (ch.mimeType.includes("image")) {
    this.session?.sendRealtimeInput({ video: ch });
    hasVideo = true;
  }

  if (hasAudio && hasVideo) {
    break;
  }
}
